\documentclass[letterpaper]{seasonalletter}
\SeasonalSetup{season=winter,weekday=friday,paper=grid,output=print,
 label={A letter on plain paper}}
\begin{document}
\begin{SeasonalLetter}
\SeasonalHeading{A lighter impression}
\SeasonalOpening{Dear friend,}
Print mode uses white paper and grey ornament. It omits background images,
grid lines and ruling. The text remains normal typeset text.

The purpose is to keep a recognisable stationery design while reducing the
amount of ink used by the background. The letter itself is unchanged.
\SeasonalClosing{With best wishes,}{A.}
\end{SeasonalLetter}
\end{document}
